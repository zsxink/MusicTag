```rust
// MusicTag — 应用配置（目录记忆，v1-ux-polish）。
//
// 配置存平台配置目录 `musictag/config.json`（dirs::config_dir）。
// 仅记录 `last_dir`（上次打开的文件目录），V1 无其他持久化需求（YAGNI）。
// 读写用 serde_json；原子写（临时文件 + rename）防写坏配置。
// 文件不存在/损坏 → 静默 None（不 panic、不污染启动）。

use std::fs;
use std::path::{Path, PathBuf};
use std::io::Write;

/// 默认配置路径：`{config_dir}/musictag/config.json`；无 config_dir（罕见）回退当前目录。
pub fn default_config_path() -> PathBuf {
    dirs::config_dir()
        .unwrap_or_else(|| PathBuf::from("."))
        .join("musictag")
        .join("config.json")
}

/// 读 `last_dir`；文件不存在 / JSON 损坏 / 目录已被删除 → `None`（静默降级）。
pub fn load_last_dir(path: &Path) -> Option<String> {
    let text = fs::read_to_string(path).ok()?;
    let cfg: serde_json::Value = serde_json::from_str(&text).ok()?;
    let dir = cfg.get("last_dir")?.as_str()?;
    let p = PathBuf::from(dir);
    if p.is_dir() { Some(dir.to_string()) } else { None }
}

/// 写 `last_dir`：原子写（临时文件 + rename），失败返回中文 Err。
pub fn save_last_dir(path: &Path, dir: &str) -> Result<(), String> {
    let json = serde_json::json!({ "last_dir": dir }).to_string();
    let parent = path.parent().unwrap_or_else(|| Path::new("."));
    fs::create_dir_all(parent).map_err(|e| format!("创建配置目录失败: {e}"))?;
    let tmp = path.with_extension("json.tmp");
    {
        let mut f = fs::File::create(&tmp).map_err(|e| format!("写配置失败: {e}"))?;
        f.write_all(json.as_bytes()).map_err(|e| format!("写配置失败: {e}"))?;
    }
    fs::rename(&tmp, path).map_err(|e| format!("写配置失败: {e}"))?;
    Ok(())
}
```

```rust
use crate::service::config;
use rfd::FileDialog;

#[tauri::command]
pub fn pick_folder() -> Option<String> {
    let mut dialog = FileDialog::new();
    // 有上次目录 → 选择器默认定位到它；首次运行 → 系统默认位置
    if let Some(dir) = config::load_last_dir(&config::default_config_path()) {
        dialog = dialog.set_directory(dir);
    }
    dialog.pick_folder().map(|dir| dir.to_string_lossy().into_owned())
}

/// 读上次打开的文件目录（目录已删除 → None，前端回退空态）。
#[tauri::command]
pub fn get_last_dir() -> Option<String> {
    config::load_last_dir(&config::default_config_path())
}

/// 记录当前打开的文件目录（每次 activateFolder 成功时调用）。
#[tauri::command]
pub fn save_last_dir(dir: String) {
    let _ = config::save_last_dir(&config::default_config_path(), &dir); // fire-and-forget，失败静默
}
```

```ts
/**
 * 启动自动加载上次目录（v1-ux-polish）：非空目录 → 复用 activateFolder 激活链路
 * （列表加载/搜索重置同一语义）；null → 保持「未打开文件夹」空态。
 * dir 由调用方（SongList onMounted）从 getLastDir() 获取后传入，避免 store 依赖 IPC 类型。
 */
export async function initLastDir(
  dir: string | null,
  loadSongs: (dir: string) => Promise<SongSummary[]>,
): Promise<void> {
  if (dir === null) return
  await activateFolder(dir, loadSongs)
}

/**
 * 记住当前目录（v1-ux-polish）：activateFolder 成功后调用，写 config.json。
 * saveDir 由调用方注入（默认 api/songs.ts saveLastDir），测试可注入桩。
 */
export async function rememberLastDir(
  dir: string,
  saveDir: (dir: string) => Promise<void>,
): Promise<void> {
  try {
    await saveDir(dir)
  } catch {
    // fire-and-forget：记忆失败不影响主流程
  }
}
```
